1   Unit Test  —  src/tests/child/gamification.spec.js

Unit Test · Vitest
What it tests: The gamification.js utility — all 6 badge conditions, the levelFor() level calculator, and badgeProgress() for live pending-point display. Covers edge cases: ISO datetime completedAt, missing child fields, and the rule that submitted (unverified) homework does not unlock badges.

Code

// src/tests/child/gamification.spec.js
import { describe, it, expect } from 'vitest'
import { earnedBadges, badgeProgress, levelFor, BADGES } from '@/utils/gamification'

const child0   = { points: 0 }
const child100 = { points: 100 }

const verifiedHw = (overrides = {}) => ({
  id: 'h_test', status: 'verified', subject: 'Math',
  dueDate: '2026-10-10', completedAt: '2026-10-08', points: 10, ...overrides,
})
const submittedHw = (overrides = {}) => ({ ...verifiedHw(overrides), status: 'submitted' })

describe('levelFor', () => {
  it('starts at level 1 with 0 points', () =>
    expect(levelFor(0)).toMatchObject({ level: 1, progress: 0, starsToNext: 50 }))
  it('reaches level 2 at 50 points', () =>
    expect(levelFor(50)).toMatchObject({ level: 2, progress: 0, starsToNext: 50 }))
  it('tracks progress within a level (25 of 50 = 50%)', () =>
    expect(levelFor(75)).toMatchObject({ level: 2, progress: 50, starsToNext: 25 }))
  it('reaches level 5 (full cake) at 200 points', () =>
    expect(levelFor(200)).toMatchObject({ level: 5, progress: 0 }))
})

describe('earnedBadges', () => {
  it('returns no badges for a new child', () =>
    expect(earnedBadges(child0, [])).toHaveLength(0))
  it('awards first-quest after one verified homework', () =>
    expect(earnedBadges(child0, [verifiedHw()]).map(b => b.id)).toContain('first-quest'))
  it('awards century at 100 points', () =>
    expect(earnedBadges(child100, []).map(b => b.id)).toContain('century'))
  it('does not award century below 100 points', () =>
    expect(earnedBadges({ points: 99 }, []).map(b => b.id)).not.toContain('century'))

  it('awards streak-3 for 3 consecutive on-time submissions', () => {
    const hw = [
      verifiedHw({ id: 'h1', completedAt: '2026-10-01', dueDate: '2026-10-02' }),
      verifiedHw({ id: 'h2', completedAt: '2026-10-03', dueDate: '2026-10-04' }),
      verifiedHw({ id: 'h3', completedAt: '2026-10-05', dueDate: '2026-10-06' }),
    ]
    expect(earnedBadges(child0, hw).map(b => b.id)).toContain('streak-3')
  })
  it('does not award streak-3 when a late submission breaks the chain', () => {
    const hw = [
      verifiedHw({ id: 'h1', completedAt: '2026-10-01', dueDate: '2026-10-02' }),
      verifiedHw({ id: 'h2', completedAt: '2026-10-05', dueDate: '2026-10-03' }), // late
      verifiedHw({ id: 'h3', completedAt: '2026-10-06', dueDate: '2026-10-07' }),
    ]
    expect(earnedBadges(child0, hw).map(b => b.id)).not.toContain('streak-3')
  })
  // ... (bookworm, rising-star, and ISO datetime edge cases omitted for brevity — see full file)
})

describe('earnedBadges requires verified status — submitted homework does not count', () => {
  it('does not award first-quest for a submitted (unverified) homework', () =>
    expect(earnedBadges(child0, [submittedHw()]).map(b => b.id)).not.toContain('first-quest'))
  it('does not award quest-champion for 5 submitted (unverified) homework', () => {
    const hw = Array.from({ length: 5 }, (_, i) => submittedHw({ id: `h${i}` }))
    expect(earnedBadges(child0, hw).map(b => b.id)).not.toContain('quest-champion')
  })
  it('awards first-quest once the homework is verified', () =>
    expect(earnedBadges(child0, [verifiedHw()]).map(b => b.id)).toContain('first-quest'))
})

describe('badgeProgress counts submitted homework for live display', () => {
  it('century counts pending points from submitted homework', () => {
    const { current, target } = badgeProgress({ id: 'century' }, { points: 90 }, [submittedHw({ id: 'h1', points: 10 })])
    expect(current).toBe(100)
    expect(target).toBe(100)
  })
})

2   Component Test 1  —  src/components/__tests__/HomeworkQuestCard.spec.js

Component Test · Vitest + Vue Test Utils
What it tests: The HomeworkQuestCard component in isolation. Mounts the component with Vue Test Utils and checks that the correct buttons appear for each role and status, that click events emit the right payload, and that the volunteer verify button is correctly disabled and shows a spinner when busy=true.

Code

// src/components/__tests__/HomeworkQuestCard.spec.js
import { describe, it, expect } from 'vitest'
import { mount } from '@vue/test-utils'
import HomeworkQuestCard from '../HomeworkQuestCard.vue'

const baseHw = {
  id: 'hw1', title: 'Count to 10', subject: 'Math',
  dueDate: '2099-12-31', points: 15, details: 'Count from 1 to 10 out loud.', status: 'assigned',
}

const mountCard = (homework, mode = 'child', extra = {}) =>
  mount(HomeworkQuestCard, { props: { homework, mode, ...extra }, global: { stubs: { Teleport: true } } })

describe('HomeworkQuestCard — child mode', () => {
  it('renders the quest title', () => {
    const wrapper = mountCard(baseHw)
    expect(wrapper.find('[data-test="homework-card"]').exists()).toBe(true)
    expect(wrapper.text()).toContain('Count to 10')
  })
  it('shows the submit button when status is assigned', () =>
    expect(mountCard(baseHw).find('[data-test="submit-quest-btn"]').exists()).toBe(true))
  it('emits submit with the homework id when the submit button is clicked', async () => {
    const wrapper = mountCard(baseHw)
    await wrapper.find('[data-test="submit-quest-btn"]').trigger('click')
    expect(wrapper.emitted('submit')?.[0]).toEqual(['hw1'])
  })
  it('hides the submit button once the quest is submitted', () =>
    expect(mountCard({ ...baseHw, status: 'submitted' }).find('[data-test="submit-quest-btn"]').exists()).toBe(false))
})

describe('HomeworkQuestCard — volunteer mode', () => {
  it('shows the verify button when a quest is waiting to be checked', () =>
    expect(mountCard({ ...baseHw, status: 'submitted' }, 'volunteer').find('[data-test="verify-homework-btn"]').exists()).toBe(true))
  it('emits verify with the homework id when the verify button is clicked', async () => {
    const wrapper = mountCard({ ...baseHw, status: 'submitted' }, 'volunteer')
    await wrapper.find('[data-test="verify-homework-btn"]').trigger('click')
    expect(wrapper.emitted('verify')?.[0]).toEqual(['hw1'])
  })
  it('shows edit and delete buttons when a quest is still assigned', () => {
    const wrapper = mountCard(baseHw, 'volunteer')
    expect(wrapper.find('[data-test="edit-homework-btn"]').exists()).toBe(true)
    expect(wrapper.find('[data-test="delete-homework-btn"]').exists()).toBe(true)
  })
  it('disables the verify button and shows spinner when busy', () => {
    const btn = mountCard({ ...baseHw, status: 'submitted' }, 'volunteer', { busy: true }).find('[data-test="verify-homework-btn"]')
    expect(btn.attributes('disabled')).toBeDefined()
    expect(btn.find('.spinner-border').exists()).toBe(true)
  })
  it('does not disable the verify button when not busy', () =>
    expect(mountCard({ ...baseHw, status: 'submitted' }, 'volunteer', { busy: false }).find('[data-test="verify-homework-btn"]').attributes('disabled')).toBeUndefined())
})

describe('HomeworkQuestCard — undo', () => {
  it('shows the undo button when status is submitted in child mode', () =>
    expect(mountCard({ ...baseHw, status: 'submitted' }).find('[data-test="undo-quest-btn"]').exists()).toBe(true))
  it('emits unsubmit with the homework id when the undo button is clicked', async () => {
    const wrapper = mountCard({ ...baseHw, status: 'submitted' })
    await wrapper.find('[data-test="undo-quest-btn"]').trigger('click')
    expect(wrapper.emitted('unsubmit')?.[0]).toEqual(['hw1'])
  })
  it('does not show the undo button when status is verified', () =>
    expect(mountCard({ ...baseHw, status: 'verified' }).find('[data-test="undo-quest-btn"]').exists()).toBe(false))
  it('does not show the undo button in volunteer mode', () =>
    expect(mountCard({ ...baseHw, status: 'submitted' }, 'volunteer').find('[data-test="undo-quest-btn"]').exists()).toBe(false))
})

3   Component Test 2  —  src/components/__tests__/ChildAvatar.spec.js

Component Test · Vitest + Vue Test Utils
What it tests: The ChildAvatar component. Verifies that a path starting with / renders an <img> element, an emoji string renders as text with no image, and an empty string renders nothing.

Code

// src/components/__tests__/ChildAvatar.spec.js
import { describe, it, expect } from 'vitest'
import { mount } from '@vue/test-utils'
import ChildAvatar from '../ChildAvatar.vue'

describe('ChildAvatar', () => {
  it('renders an img when avatar is a file path starting with /', () => {
    const wrapper = mount(ChildAvatar, { props: { avatar: '/photos/mia.webp' } })
    const img = wrapper.find('img')
    expect(img.exists()).toBe(true)
    expect(img.attributes('src')).toBe('/photos/mia.webp')
  })

  it('renders text (emoji) when avatar does not start with /', () => {
    const wrapper = mount(ChildAvatar, { props: { avatar: '🌟' } })
    expect(wrapper.find('img').exists()).toBe(false)
    expect(wrapper.text()).toBe('🌟')
  })

  it('renders nothing visible when avatar is an empty string', () => {
    const wrapper = mount(ChildAvatar, { props: { avatar: '' } })
    expect(wrapper.find('img').exists()).toBe(false)
    expect(wrapper.text()).toBe('')
  })
})

4   E2E Test 1  —  e2e/child.spec.js

E2E Test · Playwright
What it tests: The full child quest journey against a running app and real MongoDB. Tests use data-test="..." stable selectors. Two tests are state-mutating (submit + verify write to MongoDB) and are marked chromium-only so the mobile project does not mutate the database a second time.

Code

// e2e/child.spec.js  — Owner: Jachin
// Seed: child@kidslapis.sg = Ethan (c_1); h_1 is 'assigned', h_2 is 'verified'
import { test, expect } from '@playwright/test'
import { loginAs } from './helpers'

test.describe('Child quests', () => {
  test.beforeEach(async ({ page }) => {
    await loginAs(page, 'child')
    await expect(page).toHaveURL(/\/child$/)
  })

  test('My Quests page shows the tab bar, quests and completed toggle', async ({ page }) => {
    await expect(page.getByRole('tab', { name: /Quests/i })).toBeVisible()
    await expect(page.getByRole('button', { name: /Completed quests/i })).toBeVisible()
    await expect(page.locator('[data-test="homework-card"]').first()).toBeVisible()
  })

  // Mutates h_1 from 'assigned' → 'submitted'; chromium-only
  test("clicking I'm done! submits the quest and shows it as waiting for check", async ({ page }, testInfo) => {
    test.skip(testInfo.project.name === 'mobile', 'state-mutating test runs on chromium only')
    await expect(page.locator('[data-test="submit-quest-btn"]').first()).toBeVisible()
    const todoBefore = await page.locator('[data-test="submit-quest-btn"]').count()
    await page.locator('[data-test="submit-quest-btn"]').first().click()
    await expect(page.locator('[data-test="submit-quest-btn"]')).toHaveCount(todoBefore - 1)
    await expect(page.getByText('Done - waiting for check').first()).toBeVisible()
  })

  test('badge shelf renders all expected badges', async ({ page }) => {
    await page.getByRole('tab', { name: /Kueh Tin/i }).click()
    await expect(page.getByText('First Quest')).toBeVisible()
    await expect(page.getByText('Century')).toBeVisible()
  })
})

test.describe('Quest verification (volunteer)', () => {
  // Depends on the submit test above having run first (chromium-only)
  test('verifying a submitted quest removes the verify button', async ({ page }, testInfo) => {
    test.skip(testInfo.project.name === 'mobile', 'state-mutating — runs on chromium only')
    await loginAs(page, 'volunteer')
    await page.goto('/children/c_1')
    const verifyBtn = page.locator('[data-test="verify-homework-btn"]').first()
    await expect(verifyBtn).toBeVisible()
    const countBefore = await page.locator('[data-test="verify-homework-btn"]').count()
    await verifyBtn.click()
    await expect(page.locator('[data-test="verify-homework-btn"]')).toHaveCount(countBefore - 1)
  })
})

test.describe('ChildHomeView API failure', () => {
  test('shows error state when the homework API returns 500', async ({ page }) => {
    await page.route('**/homework*', (route) =>
      route.fulfill({ status: 500, contentType: 'application/json', body: JSON.stringify({ message: 'Database error' }) }))
    await loginAs(page, 'child')
    await expect(page.locator('[role="alert"]')).toBeVisible()
    await expect(page.locator('[data-test="homework-card"]')).toHaveCount(0)
  })
})

5   E2E Test 2  —  e2e/leaderboard.spec.js

E2E Test · Playwright
What it tests: The weekly leaderboard page — that the podium renders with at least one real entry from the database, and that the footer navigation link takes the child back to their home page. Both are read-only tests (no DB writes) and run on both chromium and mobile.

Code

// e2e/leaderboard.spec.js  — Owner: Jachin
// Seed: multiple children have verified homework, so podium-1 is always occupied.
import { test, expect } from '@playwright/test'
import { loginAs } from './helpers'

test.describe('Leaderboard', () => {
  test.beforeEach(async ({ page }) => {
    await loginAs(page, 'child')
    await page.goto('/child/leaderboard')
    await expect(page.getByRole('heading', { name: /Leaderboard/i })).toBeVisible()
  })

  // ── E2E Test 1: podium structure ──
  test('podium renders with at least one ranked entry', async ({ page }) => {
    // Seed gives multiple children verified stars → position 1 must be occupied.
    await expect(page.locator('[data-test="podium-1"]')).toBeVisible()
    const stars = page.locator('[data-test="podium-1"] .lb-stars')
    await expect(stars).toBeVisible()
    const text = await stars.textContent()
    expect(Number(text?.replace(/[^\d]/g, ''))).toBeGreaterThan(0)
  })

  // ── E2E Test 2: footer navigation ──
  test('footer link navigates back to the child home page', async ({ page }) => {
    const goLink = page.getByRole('link', { name: /Go do a quest/i })
    await expect(goLink).toBeVisible()
    await goLink.click()
    await expect(page).toHaveURL(/\/child$/)
  })
})
